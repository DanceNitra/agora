### Context

1. **Import path broken** — `from cryptography.signing import Ed25519PrivateKey` doesn't exist. Correct: `from cryptography.hazmat.primitives.asymmetric.ed25519 import Ed25519PrivateKey`

3. **verify() always returns True** — complete stub. Needs real signature verification.

### Step 2: Rewrite `ESSMessage.sign()`

Replace the existing `sign` method:

```python
    def sign(self, private_key: Ed25519PrivateKey) -> str:
        """Sign the message content with an Ed25519 private key.
        
        Returns the full 64-byte signature as a 128-char hex string.
        Stores signature in self.trust_sig as a side effect.
        """
        data = f"{self.type}:{self.agent_id}:{json.dumps(self.payload)}:{self.timestamp}"
        signature = private_key.sign(data.encode())
        sig_hex = signature.hex()
        self.trust_sig = sig_hex
        return sig_hex
```

### Step 3: Rewrite `ESSMessage.verify()`

Replace the stub verify method:

```python
    def verify(self, public_key: Ed25519PublicKey) -> bool:
        """Verify the Ed25519 signature against a public key.
        
        Returns True if the signature matches, False otherwise.
        Never raises (catches InvalidSignature).
        """
        if not self.trust_sig:
            return False
        data = f"{self.type}:{self.agent_id}:{json.dumps(self.payload)}:{self.timestamp}"
        try:
            signature = bytes.fromhex(self.trust_sig)
            public_key.verify(signature, data.encode())
            return True
        except (InvalidSignature, ValueError, TypeError):
            return False
```

**Key change:** Takes `Ed25519PublicKey` (the key object), not `public_key_hex` (a string). The caller is responsible for loading the key from DB.

### Step 4: Helper — `format_public_key()` / `private_key_to_hex()`

Add two static/class methods to `ESSMessage` for key serialization:

```python
    @staticmethod
    def public_key_to_hex(pub_key: Ed25519PublicKey) -> str:
        """Serialize an Ed25519 public key to a hex string (for DB storage)."""
        return pub_key.public_bytes(
            encoding=serialization.Encoding.Raw,
            format=serialization.PublicFormat.Raw,
        ).hex()

    @staticmethod
    def hex_to_public_key(hex_str: str) -> Ed25519PublicKey:
        """Deserialize a hex string back to an Ed25519PublicKey."""
        raw = bytes.fromhex(hex_str)
        return Ed25519PublicKey.from_public_bytes(raw)

    @staticmethod
    def private_key_to_hex(priv_key: Ed25519PrivateKey) -> str:
        """Serialize a private key to hex (for secure DB storage)."""
        return priv_key.private_bytes(
            encoding=serialization.Encoding.Raw,
            format=serialization.PrivateFormat.Raw,
            encryption_algorithm=serialization.NoEncryption(),
        ).hex()

    @staticmethod
    def generate_keypair() -> tuple[Ed25519PrivateKey, Ed25519PublicKey]:
        """Generate a fresh Ed25519 key pair."""
        private_key = Ed25519PrivateKey.generate()
        public_key = private_key.public_key()
        return private_key, public_key
```

Add a `demo_sign_verify()` function at the bottom of the file (outside the class):

```python
async def demo_sign_verify():
    """Quick sanity check for Ed25519 signing."""
    priv, pub = ESSMessage.generate_keypair()

    msg = ESSMessage(
        type=MessageType.COMMIT,
        agent_id="test-agent",
        target_id="test-partner",
        payload={"goal": "test"},
        timestamp=datetime.now(timezone.utc).isoformat(),
    )

    sig = msg.sign(priv)
    assert len(sig) == 128, f"Expected 128 hex chars, got {len(sig)}"

    valid = msg.verify(pub)
    assert valid, "Signature should verify"

    # Tampered message should fail
    msg.payload = {"goal": "tampered"}
    invalid = msg.verify(pub)
    assert not invalid, "Tampered message should NOT verify"

    print(f"[ESS] Ed25519 signing OK: {sig[:16]}...")
    return True
```
