### Layer 1 — Per-Agent Memory System (3 memory types)

#### New Tables — `agent_memories`

```sql
CREATE TABLE IF NOT EXISTS agent_memories (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    npc_id          TEXT NOT NULL REFERENCES dungeon_npcs(npc_id) ON DELETE CASCADE,
    memory_type     TEXT NOT NULL CHECK (memory_type IN ('episodic', 'semantic', 'procedural')),
    content         TEXT NOT NULL,
    importance      REAL NOT NULL DEFAULT 0.5 CHECK (importance >= 0 AND importance <= 1),
    emotional_tag   TEXT DEFAULT 'neutral',  -- happy, sad, angry, surprised, trusted, betrayed
    source          TEXT DEFAULT 'experience',  -- experience, conversation, brainstorm, self_reflection
    related_npc_id  TEXT,  -- which other NPC this memory is about (optional)
    decay_factor    REAL NOT NULL DEFAULT 1.0 CHECK (decay_factor >= 0 AND decay_factor <= 1),
    created_at      TEXT NOT NULL DEFAULT (datetime('now')),
    last_recalled_at TEXT
);
CREATE INDEX IF NOT EXISTS idx_memories_npc ON agent_memories(npc_id, memory_type);
CREATE INDEX IF NOT EXISTS idx_memories_importance ON agent_memories(npc_id, importance DESC);
```

**Three memory types:**
- **episodic** — "I saw Lyra in the Library at noon. She was carrying a scroll." (specific events, time-stamped, can decay)
- **semantic** — "The Library contains ancient runes. Lyra is good at cartography." (general knowledge, slow decay)
- **procedural** — "To forge a key: first collect iron ore, then heat in forge, then hammer into shape." (skill knowledge, no decay)

#### MemoryAgent class — new file `agent_os/memory_agent.py`

```python
"""
Per-agent memory system with importance scoring, decay, emotional tagging,
and multi-type retrieval (episodic/semantic/procedural).

Each agent has an independent memory store that persists in SQLite.
Memories decay over time unless they are important or frequently recalled.
"""
import json
import math
import random
from datetime import datetime, timezone
from typing import Optional

class MemoryAgent:
    """Individual memory system for one NPC."""

    DECAY_RATE = 0.05       # loss per day (simulated per tick)
    IMPORTANCE_THRESHOLD = 0.8  # memories above this survive pruning
    MAX_EPISODIC = 100       # max episodic memories per agent (prune oldest)
    MAX_SEMANTIC = 50        # max semantic memories
    MAX_PROCEDURAL = 30      # max procedural (rarely added)

    def __init__(self, db, npc_id: str):
        self.db = db
        self.npc_id = npc_id

    async def store_memory(
        self,
        content: str,
        memory_type: str = "episodic",
        importance: float = 0.5,
        emotional_tag: str = "neutral",
        source: str = "experience",
        related_npc_id: Optional[str] = None,
    ) -> int:
        """Store a new memory. Returns memory ID."""
        importance = max(0.0, min(1.0, importance))
        cursor = await self.db.execute(
            "INSERT INTO agent_memories (npc_id, memory_type, content, importance, "
            "emotional_tag, source, related_npc_id, decay_factor, last_recalled_at) "
            "VALUES (?, ?, ?, ?, ?, ?, ?, 1.0, datetime('now'))",
            (self.npc_id, memory_type, content, importance, emotional_tag,
             source, related_npc_id),
        )
        await self.db.commit()
        mem_id = cursor.lastrowid

        # Prune if over capacity
        await self._prune_if_needed(memory_type)
        return mem_id

    async def recall(
        self,
        query: str = "",
        memory_type: Optional[str] = None,
        limit: int = 5,
        min_importance: float = 0.0,
        with_emotion: Optional[str] = None,
        about_npc: Optional[str] = None,
    ) -> list[dict]:
        """Recall memories matching the query. Uses keyword overlap scoring.
        
        Scoring formula:
            score = importance * 0.6 + keyword_overlap * 0.3 + recency_factor * 0.1
        
        Returns top `limit` memories sorted by score descending.
        """
        conditions = ["npc_id=?"]
        params = [self.npc_id]

        if memory_type:
            conditions.append("memory_type=?")
            params.append(memory_type)
        if min_importance > 0:
            conditions.append("importance>=?")
            params.append(min_importance)
        if with_emotion:
            conditions.append("emotional_tag=?")
            params.append(with_emotion)
        if about_npc:
            conditions.append("related_npc_id=?")
            params.append(about_npc)

        cursor = await self.db.execute(
            f"SELECT * FROM agent_memories WHERE {' AND '.join(conditions)} "
            f"ORDER BY importance DESC, last_recalled_at DESC LIMIT ?",
            params + [limit * 3],  # fetch extra for scoring
        )
        rows = await cursor.fetchall()

        if not rows:
            return []

        # Score each memory
        query_keywords = set(
            w.lower() for w in query.split() if len(w) > 3
        ) if query else set()

        scored = []
        for row in rows:
            mem = dict(row)
            now = datetime.now(timezone.utc)
            last_recall = mem.get("last_recalled_at")
            if last_recall:
                try:
                    if isinstance(last_recall, str):
                        last_dt = datetime.fromisoformat(last_recall)
                    else:
                        last_dt = last_recall
                    hours_ago = (now - last_dt).total_seconds() / 3600
                    recency_factor = math.exp(-hours_ago / 24)  # decays over 24h
                except Exception:
                    recency_factor = 0.5
            else:
                recency_factor = 0.3

            mem_content = mem.get("content", "").lower()
            keyword_overlap = 0.0
            if query_keywords and mem_content:
                overlap = sum(1 for kw in query_keywords if kw in mem_content)
                keyword_overlap = min(1.0, overlap / max(len(query_keywords), 1))

            score = (
                mem["importance"] * 0.6
                + keyword_overlap * 0.3
                + recency_factor * 0.1
            )
            scored.append((score, mem))

        scored.sort(key=lambda x: x[0], reverse=True)

        # Update last_recalled_at for retrieved memories
        result = scored[:limit]
        for score, mem in result:
            await self.db.execute(
                "UPDATE agent_memories SET last_recalled_at=datetime('now'), "
                "decay_factor=MIN(1.0, decay_factor+0.1) WHERE id=?",
                (mem["id"],),
            )
        await self.db.commit()

        return [mem for score, mem in result]

    async def remember_recent(self, limit: int = 10) -> list[dict]:
        """Get most recent memories (for LLM context injection)."""
        cursor = await self.db.execute(
            "SELECT * FROM agent_memories WHERE npc_id=? ORDER BY created_at DESC LIMIT ?",
            (self.npc_id, limit),
        )
        return [dict(r) for r in await cursor.fetchall()]

    async def get_emotional_summary(self) -> dict:
        """Get emotional state summary from recent memories."""
        cursor = await self.db.execute(
            "SELECT emotional_tag, COUNT(*) as cnt FROM agent_memories "
            "WHERE npc_id=? AND emotional_tag!='neutral' "
            "GROUP BY emotional_tag ORDER BY cnt DESC",
            (self.npc_id,),
        )
        rows = await cursor.fetchall()
        return {r["emotional_tag"]: r["cnt"] for r in rows}

    async def _prune_if_needed(self, memory_type: str):
        """Prune oldest memories if over capacity."""
        max_map = {"episodic": self.MAX_EPISODIC, "semantic": self.MAX_SEMANTIC,
                    "procedural": self.MAX_PROCEDURAL}
        max_count = max_map.get(memory_type, 100)

        cursor = await self.db.execute(
            "SELECT COUNT(*) as c FROM agent_memories WHERE npc_id=? AND memory_type=? "
            "AND importance < ?",
            (self.npc_id, memory_type, self.IMPORTANCE_THRESHOLD),
        )
        row = await cursor.fetchone()
        low_imp_count = row["c"] if row else 0

        cursor = await self.db.execute(
            "SELECT COUNT(*) as c FROM agent_memories WHERE npc_id=? AND memory_type=?",
            (self.npc_id, memory_type),
        )
        row = await cursor.fetchone()
        total = row["c"] if row else 0

        if total > max_count and low_imp_count > 0:
            # Delete oldest low-importance memories
            prune_count = total - max_count
            await self.db.execute(
                "DELETE FROM agent_memories WHERE id IN ("
                "SELECT id FROM agent_memories WHERE npc_id=? AND memory_type=? "
                "AND importance < ? ORDER BY last_recalled_at ASC LIMIT ?"
                ")",
                (self.npc_id, memory_type, self.IMPORTANCE_THRESHOLD, prune_count),
            )
            await self.db.commit()

    async def decay_all(self):
        """Apply decay to all memories. Called once per tick."""
        await self.db.execute(
            "UPDATE agent_memories SET decay_factor = MAX(0.0, decay_factor - ?) "
            "WHERE npc_id=? AND memory_type IN ('episodic', 'semantic') "
            "AND decay_factor > 0",
            (self.DECAY_RATE, self.npc_id),
        )
        # Forgetting: remove memories where decay_factor <= 0 AND importance < threshold
        await self.db.execute(
            "DELETE FROM agent_memories WHERE npc_id=? AND decay_factor <= 0 "
            "AND importance < ?",
            (self.npc_id, self.IMPORTANCE_THRESHOLD),
        )
        await self.db.commit()

    async def get_relevant_memories_for_prompt(self, context: str, max_memories: int = 8) -> str:
        """Get a formatted string of relevant memories for LLM prompt injection.
        
        Returns something like:
        ```
        --- Your Memories ---
        [episodic][0.85] You saw Lyra in the Library carrying a scroll. (2h ago)
        [semantic][0.72] Lyra is skilled at cartography.
        [procedural][0.95] To forge a key: collect ore → heat → hammer.
        ```
        """
        recalled = await self.recall(context, limit=max_memories)
        if not recalled:
            # Fallback: just recent memories
            recalled = await self.remember_recent(limit=3)

        lines = ["--- Your Memories ---"]
        for mem in recalled:
            mem_type = mem["memory_type"][:4]
            importance = mem["importance"]
            content = mem["content"][:150]
            emotion = mem["emotional_tag"]
            emotion_tag = f" [{emotion}]" if emotion != "neutral" else ""
            lines.append(
                f"  [{mem_type}] [{importance:.2f}] {content}{emotion_tag}"
            )
        return "\n".join(lines)
```

#### New method — `_start_conversation()` in AgentOS

```python
async def _start_conversation(
    self,
    speaker_id: str,
    target_id: str,
    topic: str,
    intent: str = "chat",
    broadcast_fn=None,
) -> str:
    """Start a conversation between two agents.
    
    Both agents call LLM in sequence (speaker first, then target responds).
    Returns the conversation session_id for continuation.
    """
    session_id = str(uuid.uuid4())
    speaker_name = await self._get_npc_name(speaker_id)
    target_name = await self._get_npc_name(target_id)

    # ── Turn 1: Speaker initiates ──
    speaker_prompt = (
        f"You are {speaker_name}. You want to talk to {target_name} about '{topic}' "
        f"(intent: {intent}). "
        f"Write what you say — a natural {intent} message, 1-2 sentences. "
        f"Be in-character with your personality. "
        f"Respond with JSON: {{\"message\": \"...\", \"feeling\": \"...\"}}"
    )
    speaker_decision = await asyncio.to_thread(
        dungeon_agent_think, speaker_name, "conversation", speaker_prompt, "cheap",
    )
    speaker_msg = speaker_decision.get("message", f"Hey {target_name}, let's talk about {topic}.")
    speaker_feeling = speaker_decision.get("feeling", "curious")

    # Store speaker's turn
    await self.db.execute(
        "INSERT INTO agent_conversations (session_id, speaker_id, target_id, message, "
        "intent, turn_number, speaker_name, target_name) "
        "VALUES (?, ?, ?, ?, ?, 1, ?, ?)",
        (session_id, speaker_id, target_id, speaker_msg, intent, speaker_name, target_name),
    )

    # Store as memory for both agents
    mem_agent_speaker = MemoryAgent(self.db, speaker_id)
    await mem_agent_speaker.store_memory(
        f"You started a conversation with {target_name} about '{topic}': \"{speaker_msg[:100]}\"",
        memory_type="episodic", importance=0.6, emotional_tag=speaker_feeling,
        source="conversation", related_npc_id=target_id,
    )

    # ── Turn 2: Target responds ──
    # Inject the speaker's message + target's memory of the speaker
    target_memory_agent = MemoryAgent(self.db, target_id)
    target_memories = await target_memory_agent.recall(
        speaker_name, limit=3, min_importance=0.3,
    )
    target_mem_str = ""
    if target_memories:
        target_mem_str = "\n".join(
            f"  - {m['content'][:100]}" for m in target_memories
        )

    target_prompt = (
        f"You are {target_name}. {speaker_name} just said to you: \"{speaker_msg}\"\n"
        f"They started a conversation about '{topic}'.\n"
        f"{'Your memories of ' + speaker_name + ':' + target_mem_str if target_mem_str else ''}\n"
        f"Respond naturally. Be in-character. 1-2 sentences. "
        f"Respond with JSON: {{\"message\": \"...\", \"feeling\": \"...\"}}"
    )
    target_decision = await asyncio.to_thread(
        dungeon_agent_think, target_name, "conversation", target_prompt, "cheap",
    )
    target_msg = target_decision.get("message", f"Interesting topic, {speaker_name}!")
    target_feeling = target_decision.get("feeling", "curious")

    # Store target's turn
    await self.db.execute(
        "INSERT INTO agent_conversations (session_id, speaker_id, target_id, message, "
        "intent, turn_number, speaker_name, target_name) "
        "VALUES (?, ?, ?, ?, ?, 2, ?, ?)",
        (session_id, target_id, speaker_id, target_msg, intent, target_name, speaker_name),
    )

    # Store as memory for target
    await target_memory_agent.store_memory(
        f"{speaker_name} talked to you about '{topic}' and said: \"{speaker_msg[:80]}\" "
        f"You replied: \"{target_msg[:80]}\"",
        memory_type="episodic", importance=0.6, emotional_tag=target_feeling,
        source="conversation", related_npc_id=speaker_id,
    )

    await self.db.commit()

    # Broadcast the conversation
    if broadcast_fn:
        await broadcast_fn("agent_conversation", {
            "session_id": session_id,
            "speaker": speaker_name,
            "target": target_name,
            "topic": topic,
            "turns": [
                {"name": speaker_name, "message": speaker_msg, "feeling": speaker_feeling},
                {"name": target_name, "message": target_msg, "feeling": target_feeling},
            ],
        })

    return session_id


async def _continue_conversation(
    self,
    session_id: str,
    speaker_id: str,
    target_id: str,
    broadcast_fn=None,
) -> bool:
    """Continue a conversation by one more turn pair.
    
    Returns False if both agents want to end the conversation.
    """
    # Get last 2 messages
    cursor = await self.db.execute(
        "SELECT * FROM agent_conversations WHERE session_id=? ORDER BY turn_number DESC LIMIT 2",
        (session_id,),
    )
    turns = await cursor.fetchall()
    if not turns:
        return False

    current_turn = turns[0]["turn_number"]
    speaker_name = await self._get_npc_name(speaker_id)
    target_name = await self._get_npc_name(target_id)

    # Check if agent wants to continue
    continue_check_prompt = (
        f"You are {speaker_name}. You've been talking with {target_name} "
        f"(current turn {current_turn}). "
        f"The last thing said was: \"{turns[0]['message'][:100]}\"\n"
        f"Do you want to continue this conversation or end it? "
        f"Respond with JSON: {{\"continue\": true/false, \"message\": \"...\"}}"
    )
    check_decision = await asyncio.to_thread(
        dungeon_agent_think, speaker_name, "conversation", continue_check_prompt, "cheap",
    )
    if not check_decision.get("continue", True):
        return False

    # Speaker says next thing
    new_msg = check_decision.get("message", "...")
    await self.db.execute(
        "INSERT INTO agent_conversations (session_id, speaker_id, target_id, message, "
        "intent, turn_number, speaker_name, target_name) "
        "VALUES (?, ?, ?, ?, 'chat', ?, ?, ?)",
        (session_id, speaker_id, target_id, new_msg, current_turn + 1, speaker_name, target_name),
    )
    await self.db.commit()

    if broadcast_fn:
        await broadcast_fn("agent_conversation_turn", {
            "session_id": session_id,
            "speaker": speaker_name,
            "target": target_name,
            "message": new_msg,
            "turn": current_turn + 1,
        })

    return True
```

### Layer 3 — Personality Engine (Big 5 shapes LLM output)

The Big 5 personality traits (Openness, Conscientiousness, Extraversion, Agreeableness, Neuroticism) already exist in the `agent_soul` table — but they're never used. We need to:

1. **Inject them into every LLM prompt** as character behavior instructions
2. **Let them affect state transitions** (e.g., high Neuroticism → easier to panic, high Extraversion → more likely to start conversations)

#### Personality injection — in `_think()` prompt

After the context building, add:

```python
# ── Personality affect ──
personality = json.loads(npc.get("personality", "{}") or "{}")

# Build personality instruction block
personality_lines = ["--- Your Personality ---"]
if personality:
```

```python
# In the rule-based fallback, add personality modifiers:

personality = json.loads(npc.get("personality", "{}") or "{}")
neuroticism = personality.get("neuroticism", 0.5)
extraversion = personality.get("extraversion", 0.5)

# High neuroticism → easier to panic
panic_threshold = 20 + (neuroticism * 20)  # 20-40 health triggers panic
if health < panic_threshold:
    new_state = "panicked"
elif fatigue > 70 or stamina < 20:
    new_state = "resting"
# ...

# High extraversion → more likely to start conversations
if extraversion > 0.6 and new_state == "focused" and random.random() < 0.15:
    new_state = "social"  # new state — agent wants to talk
```

```python
"""
Brainstorming engine: agents generate ideas in structured sessions.

Process:
1. An agent (or orchestrator) proposes a topic
2. Multiple agents submit initial ideas (LLM)
3. Agents build on each other's ideas (LLM with context of previous ideas)
4. Agents vote on best ideas
5. Highest-scoring ideas become quests or system upgrades

import json
import random
import uuid
from typing import Optional

from agora.execution.llm_client import dungeon_agent_think


class BrainstormEngine:
    def __init__(self, db):
        self.db = db

    async def start_session(
        self, topic: str, initiator_name: str, initiator_id: str
    ) -> str:
        """Start a new brainstorming session. Returns session_id."""
        session_id = str(uuid.uuid4())
        await self.db.execute(
            "INSERT INTO agent_brainstorm_sessions (session_id, topic, initiator_id, status) "
            "VALUES (?, ?, ?, 'active')",

        return session_id

    async def generate_ideas(

        participant_ids: list[str],
        broadcast_fn=None,
    ) -> list[dict]:
        """All participants generate initial ideas on the topic.
        
        Each agent calls LLM independently. Returns list of ideas.
        """
        cursor = await self.db.execute(
            "SELECT topic FROM agent_brainstorm_sessions WHERE session_id=?",
            (session_id,),
        )
        row = await cursor.fetchone()
        if not row:
            return []
        topic = row["topic"]

        ideas = []
        for npc_id in participant_ids:
            name = await self._get_npc_name(npc_id)
            if not name:
                continue

            # Get this agent's skills/abilities for relevant ideas
            cursor_s = await self.db.execute(

            abilities_str = "; ".join(f"{a['ability_name']} ({a['power_level']}/10)" for a in abilities)

    async def build_on_ideas(

                # Find which existing idea this builds on
                builds_on_id = None
                for existing in existing_ideas:
                    if builds_on_text and existing["idea_content"][:30] in builds_on_text:
                        builds_on_id = existing["id"]
                        break

                if idea_content:

                        "idea_type, builds_on_id, impact_score, feasibility) "

                         float(decision.get("feasibility", 0.5))),
                    )
                    new_builds.append({
                        "npc_name": name,
                        "idea": idea_content[:500],
                        "impact": float(decision.get("impact", 0.5)),
                        "feasibility": float(decision.get("feasibility", 0.5)),

            except Exception as e:

        if broadcast_fn:
            await broadcast_fn("brainstorm_builds", {

    async def vote_on_ideas(self, session_id: str, voter_ids: list[str]) -> list[dict]:
        """Agents vote on the best ideas.
        
        Each agent picks their top 3 ideas. Returns sorted list.
        """
        cursor = await self.db.execute(
            "SELECT * FROM agent_brainstorm_ideas WHERE session_id=? ORDER BY created_at",
            (session_id,),
        )
        ideas = [dict(r) for r in await cursor.fetchall()]
        if not ideas:
            return []

        ideas_summary = "\n".join(
            f"  {i['id']}. \"{i['idea_content'][:100]}\""
            for i in ideas
        )

        all_votes = []
        for npc_id in voter_ids:
            name = await self._get_npc_name(npc_id)
            if not name:
                continue

            prompt = (
                f"You are {name}. Vote on the best ideas from this brainstorming session:\n"
                f"{ideas_summary}\n\n"
                f"Pick your TOP 3 ideas by number. "
                f"Respond with JSON: {{\"votes\": [1, 2, 3], \"reason\": \"...\"}}"
            )
            try:
                import asyncio
                decision = await asyncio.to_thread(
                    dungeon_agent_think, name, "brainstorm", prompt, "cheap",
                )
                votes = decision.get("votes", [])
                for idea_id in votes:
                    if any(i["id"] == idea_id for i in ideas):
                        all_votes.append(idea_id)
            except Exception:
                pass

        # Tally votes
        from collections import Counter
        tally = Counter(all_votes)
        for idea in ideas:
            idea["votes"] = tally.get(idea["id"], 0)

        ideas.sort(key=lambda x: x["votes"], reverse=True)
        return ideas

    async def complete_session(self, session_id: str):

        return row["npc_name"] if row else None
```

```python
async def _query_collective(self, query: str, limit: int = 5) -> list[dict]:
    """Search the collective knowledge pool. Keyword-based."""
    keywords = [w for w in query.lower().split() if len(w) > 3]
    if not keywords:
        return []

    # Build LIKE conditions
    conditions = " OR ".join(
        "(LOWER(title) LIKE ? OR LOWER(content) LIKE ?)" for _ in keywords
    )
    params = []
    for kw in keywords:
        params.extend([f"%{kw}%", f"%{kw}%"])

    cursor = await self.db.execute(
        f"SELECT * FROM collective_knowledge WHERE {conditions} "
        f"ORDER BY verification_count DESC, confidence DESC LIMIT ?",
        params + [limit],
    )
    return [dict(r) for r in await cursor.fetchall()]
```

```python
async def _agentic_os_cycle(app: FastAPI):

    Phase 1: Body update (stamina, fatigue, hunger)
    Phase 2: Perception (what's nearby, stigmergy signals, collective knowledge)
    Phase 3: Memory recall (relevant memories injected into prompt)
    Phase 4: Think (LLM with full context: body + personality + memories + trust + nearby)
    Phase 5: Execute (LLM decision → visible action + ESS recording)
    Phase 6: Memory consolidation (store episodic memory of what happened)
    Phase 7: Broadcast (WebSocket push for UI)

    db = app.state.db

                        "the dungeon's secrets", "a new discovery",
                        "how to improve cooperation", "ancient artifacts",
                        "a trading opportunity", "a threat in the dungeon",
                    ]
                    topic = random.choice(topic_choices)
                    await agent_os._start_conversation(
                        npc_id, other["npc_id"], topic,
                        intent="chat",
                    )
            except Exception:
                pass

        # Phase 6b: Brainstorm trigger (every 10 ticks, 1 agent initiates)
        if (
            tick_count % 10 == 0
            and npc_id == npcs[0]["npc_id"]  # only first NPC can initiate
            and random.random() < 0.3
        ):
            topics = [
                "How can we make the dungeon safer?",
                "What new resources should we seek?",
                "How can agents help each other better?",
                "What's the most valuable thing in the dungeon?",
                "How can we improve our coordination?",
            ]
            bm = BrainstormEngine(db)
            session_id = await bm.start_session(
                random.choice(topics), name, npc_id,
            )
            all_ids = [n["npc_id"] for n in npcs]
            await bm.generate_ideas(session_id, all_ids)
            await bm.build_on_ideas(session_id, all_ids)

        # Phase 7: Broadcast state
        from agora.api.dungeon import broadcast
        await broadcast(app, "agent_cycle", {
            "agent": name,
            "state": state_of_mind,
            "health": npc["health"],
            "tick": tick_count,
        })

    # ESS epoch + checkpoint
    epoch_engine = getattr(app.state, "epoch_engine", None)
    if epoch_engine:
        await epoch_engine.tick(app)

    checkpointer = getattr(app.state, "checkpointer", None)
    if checkpointer and tick_count % 50 == 0:
        await checkpointer.checkpoint_all("trust")
```
