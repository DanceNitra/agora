### Step 2 — VaultReader

File: `server/agora/agent_os/vault_bridge/vault_reader.py`

```python
"""Vault reader — reads book concepts from the my-second-brain vault.

Two modes:
  - vault_path set: reads from ~/my-second-brain/04 Resources/Concepts/*.md
    with keyword search (FAISS RAG fallback if available)
  - vault_path empty: returns 3 built-in book concepts for testing
"""
import json
import os
import random
import subprocess
from typing import Optional


# ── Built-in mock concepts for when vault_path is not configured ──

MOCK_CONCEPTS = [
    {
        "title": "Antifragile",
        "book": "Antifragile (Taleb)",
        "text": "Systems that gain from disorder are antifragile. Via negativa: what you don't do matters more than what you do. Removing bad options improves the system more than adding good ones.",
    },
    {
        "title": "GEB — Strange Loops",
        "book": "Gödel, Escher, Bach (Hofstadter)",
        "text": "Strange loops arise when a system can represent itself. Self-reference creates emergent complexity. An agent that can model its own thinking gains meta-cognitive abilities.",
    },
    {
        "title": "Thinking in Systems",
        "book": "Thinking in Systems (Meadows)",
        "text": "Systems are made of stocks, flows, and feedback loops. The most effective interventions target system structure, not individual behavior. Leverage points range from parameters to paradigm.",
    },
]


class VaultReader:
    """Read book concepts from the vault.

    Args:
        vault_path: Path to my-second-brain repo clone.
                    If None or empty, returns MOCK_CONCEPTS (for testing).
    """

    def __init__(self, vault_path: Optional[str] = None):
        self.vault_path = vault_path
        self._concepts_dir = None
        self._rag_script = None
        self._rag_venv = None

        if vault_path:
            expanded = os.path.expanduser(vault_path)
            if os.path.isdir(expanded):
                self._concepts_dir = os.path.join(expanded, "04 Resources", "Concepts")
                self._rag_script = os.path.join(
                    expanded, "06 System", "Scripts", "vault_rag_tool.py"
                )
                self._rag_venv = os.path.join(
                    expanded, "06 System", "Scripts", ".venv-rag"
                )
                print(f"[VaultReader] Real vault: {expanded}")
            else:
                print(f"[VaultReader] Vault path '{expanded}' not found — using built-in mock")
                self.vault_path = None

    async def query(self, query: str, top_k: int = 3) -> list[dict]:
        """Search vault for concepts relevant to query.

        Priority:
          1. FAISS RAG script (if venv + index exist)
          2. Keyword search over concept .md files
          3. Built-in mock concepts (fallback)

        Returns list of {title, book, text}
        """
        # Method 1: FAISS RAG
        if self._rag_script and self._rag_venv and os.path.exists(self._rag_venv):
            try:
                result = subprocess.run(
                    f"source {self._rag_venv}/bin/activate && "
                    f"python3 {self._rag_script} \"{query}\"",
                    shell=True, capture_output=True, text=True, timeout=30,
                )
                if result.returncode == 0 and result.stdout.strip():
                    parsed = json.loads(result.stdout)
                    if parsed:
                        return [
                            {
                                "title": p.get("title", "Unknown"),
                                "book": p.get("source", "vault"),
                                "text": p.get("text", "")[:300],
                            }
                            for p in parsed[:top_k]
                        ]
            except Exception:
                pass

        # Method 2: Keyword search over concept files
        if self._concepts_dir and os.path.isdir(self._concepts_dir):
            keywords = [w.lower() for w in query.split() if len(w) > 3]
            if keywords:
                scored = []
                for fname in os.listdir(self._concepts_dir):
                    if not fname.endswith(".md"):
                        continue
                    fpath = os.path.join(self._concepts_dir, fname)
                    try:
                        with open(fpath, "r") as f:
                            content = f.read()
                    except Exception:
                        continue
                    content_lower = content.lower()
                    title = fname.replace(".md", "").replace("_", " ")
                    title_lower = title.lower()
                    score = sum(1 for kw in keywords if kw in content_lower or kw in title_lower)
                    if score > 0:
                        # Extract first meaningful paragraph (skip frontmatter)
                        body = content
                        if content.startswith("---"):
                            parts = content.split("---", 2)
                            if len(parts) >= 3:
                                body = parts[2]
                        # Take first 300 chars of non-empty lines
                        text_lines = [l.strip() for l in body.split("\n") if l.strip()]
                        preview = " ".join(text_lines[:5])[:300]
                        scored.append((score, {
                            "title": title,
                            "book": "obsidian_vault",
                            "text": preview,
                        }))
                scored.sort(key=lambda x: x[0], reverse=True)
                if scored:
                    return [s[1] for s in scored[:top_k]]

        # Method 3: Built-in mock
        keywords = [w.lower() for w in query.split() if len(w) > 3]
        scored_mock = []
        for c in MOCK_CONCEPTS:
            score = sum(1 for kw in keywords if kw in (c["title"] + c["text"] + c["book"]).lower())
            scored_mock.append((score, c))
        scored_mock.sort(key=lambda x: x[0], reverse=True)
        return [dict(c) for _, c in scored_mock[:top_k]]

    async def get_concept(self, title: str) -> Optional[str]:
        """Get full .md content of a concept by title."""
        # Try vault first
        if self._concepts_dir and os.path.isdir(self._concepts_dir):
            title_lower = title.lower().replace(" ", "_")
            for fname in os.listdir(self._concepts_dir):
                if title_lower in fname.lower().replace(".md", ""):
                    fpath = os.path.join(self._concepts_dir, fname)
                    with open(fpath, "r") as f:
                        return f.read()
        return None

    async def get_random_insight(self) -> dict:
        """Get one random insight for 'book of the tick' feature."""
        # Try vault first
        if self._concepts_dir and os.path.isdir(self._concepts_dir):
            files = [f for f in os.listdir(self._concepts_dir) if f.endswith(".md")]
            if files:
                fname = random.choice(files)
                fpath = os.path.join(self._concepts_dir, fname)
                try:
                    with open(fpath, "r") as f:
                        content = f.read()
                    body = content
                    if content.startswith("---"):
                        parts = content.split("---", 2)
                        if len(parts) >= 3:
                            body = parts[2]
                    text = " ".join(body.split("\n")[:5])[:200]
                    return {"title": fname.replace(".md", ""), "text": text, "book": "obsidian_vault"}
                except Exception:
                    pass
        # Fallback to mock
        c = random.choice(MOCK_CONCEPTS)
        return {"title": c["title"], "text": c["text"][:200], "book": c["book"]}
```

### Step 3 — VaultWriter

File: `server/agora/agent_os/vault_bridge/vault_writer.py`

```python
"""Vault writer — agents write .md notes to the vault and git push.

Two modes:
  - vault_path set: writes to ~/my-second-brain/04 Resources/Concepts/ + git commit/push
  - vault_path empty: writes to /tmp/agora-vault-output/ (no git, for testing)
"""
import os
import subprocess
from datetime import datetime
from typing import Optional


class VaultWriter:
    """Write knowledge notes to the vault.

    Args:
        vault_path: Path to my-second-brain repo clone.
                    If None or empty, writes to /tmp/ for testing.
    """

    def __init__(self, vault_path: Optional[str] = None, git_ssh_key: str = ""):
        self.vault_path = os.path.expanduser(vault_path) if vault_path else ""
        self.git_ssh_key = os.path.expanduser(git_ssh_key) if git_ssh_key else ""

        if self.vault_path and os.path.isdir(self.vault_path):
            self.concepts_dir = os.path.join(self.vault_path, "04 Resources", "Concepts")
            os.makedirs(self.concepts_dir, exist_ok=True)
            print(f"[VaultWriter] Real vault: {self.vault_path}")
        else:
            self.concepts_dir = "/tmp/agora-vault-output"
            os.makedirs(self.concepts_dir, exist_ok=True)
            self.vault_path = ""
            print(f"[VaultWriter] Test mode: {self.concepts_dir}")

    async def write_note(self, title: str, content: str, tags: list[str],
                         agent_name: str = "") -> str:
        """Write a .md note with Obsidian frontmatter. Returns file path."""
        safe_title = title.replace(" ", "_").replace("/", "_").replace(":", "_")[:80]
        fname = f"{safe_title}.md"
        fpath = os.path.join(self.concepts_dir, fname)

        frontmatter = (
            f"---\n"
            f"title: {title}\n"
            f"status: seedling\n"
            f"domain: dungeon-generated\n"
            f"tags:\n"
        )
        for tag in tags:
            frontmatter += f"  - {tag}\n"
        frontmatter += (
            f"created: {datetime.now().strftime('%Y-%m-%d')}\n"
            f"generated_by: dungeon_agent_{agent_name}\n"
            f"dungeon_session: true\n"
            f"---\n\n"
        )

        with open(fpath, "w") as f:
            f.write(frontmatter + content)

        print(f"[VaultWriter] Wrote: {fpath}")
        return fpath

    async def git_commit_and_push(self, file_path: str, message: str):
        """Git add, commit, push.

        If vault_path is set (real vault mode), actually pushes.
        In test mode (/tmp), just logs.
        """
        if not self.vault_path:
            print(f"[VaultWriter] Test mode — would git commit '{message}' for {file_path}")
            return

        vault_dir = self.vault_path
        env = os.environ.copy()
        if self.git_ssh_key and os.path.exists(self.git_ssh_key):
            env["GIT_SSH_COMMAND"] = (
                f"ssh -i {self.git_ssh_key} -o StrictHostKeyChecking=no"
            )

        # First git pull to avoid conflicts
        try:
            subprocess.run(
                ["git", "-C", vault_dir, "pull", "--rebase"],
                env=env, capture_output=True, timeout=30,
            )
        except Exception as e:
            print(f"[VaultWriter] git pull warning: {e}")

        try:
            subprocess.run(
                ["git", "-C", vault_dir, "add", file_path],
                env=env, capture_output=True, check=True, timeout=30,
            )
            subprocess.run(
                ["git", "-C", vault_dir, "commit", "-m", message],
                env=env, capture_output=True, check=True, timeout=30,
            )
            subprocess.run(
                ["git", "-C", vault_dir, "push"],
                env=env, capture_output=True, check=True, timeout=60,
            )
            print(f"[VaultWriter] Pushed: {message}")
        except subprocess.CalledProcessError as e:
            stderr = e.stderr.decode() if e.stderr else ""
            print(f"[VaultWriter] Git error (non-fatal): {stderr[:200]}")
```

"""Vault Bridge — read/write knowledge vault from dungeon agents.
