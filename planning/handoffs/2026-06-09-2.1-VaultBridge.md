```python
"""Vault reader abstraction — read knowledge from a vault.

Two strategies:
  - mock:  reads from agent_os/vault_bridge/mock_vault/ (in repo, git-tracked)
  - local: reads from real Obsidian vault via FAISS RAG or direct file access
"""
import json
import os
import random
from abc import ABC, abstractmethod
from typing import Optional


class VaultReader(ABC):
    """Abstract vault reader. 
    
    query() returns relevant concept chunks for LLM prompt injection.
    get_concept() returns the full .md content of a concept note.
    get_book_of_the_day() returns one random book insight per tick (rotating).
    """

    @abstractmethod
    async def query(self, query: str, top_k: int = 3) -> list[dict]:
        """Search vault for concepts relevant to query.
        
        Returns list of {title, text, book, score}
        """
        ...

    @abstractmethod
    async def get_concept(self, title: str) -> Optional[str]:
        """Get full .md content of a concept note by title."""
        ...

    @abstractmethod
    async def get_random_insight(self) -> dict:
        """Get one random book insight (for 'book of the tick' feature)."""
        ...


class DevMockVaultReader(VaultReader):
    """Reads from mock_vault/ directory in the agora repo.
    
    12 concept .md files, each with a core mechanism + 3 applications.
    Uses keyword matching (no embeddings) for query().
    Mock vault is git-tracked and can be evolved by committing changes.
    """

    def __init__(self, mock_dir: str = None):
        base = os.path.dirname(os.path.abspath(__file__))
        self.mock_dir = mock_dir or os.path.join(base, "mock_vault", "concepts")
        self._index = {}  # title -> {title, text, book, tags}
        self._loaded = False

    def _ensure_loaded(self):
        if self._loaded:
            return
        if not os.path.isdir(self.mock_dir):
            os.makedirs(self.mock_dir, exist_ok=True)
            self._loaded = True
            return
        for fname in os.listdir(self.mock_dir):
            if not fname.endswith(".md"):
                continue
            fpath = os.path.join(self.mock_dir, fname)
            with open(fpath, "r") as f:
                content = f.read()
            title = fname.replace(".md", "").replace("_", " ")
            # Parse frontmatter
            tags = []
            book = ""
            text = content
            if content.startswith("---"):
                parts = content.split("---", 2)
                if len(parts) >= 3:
                    text = parts[2].strip()
                    for line in parts[1].split("\n"):
                        line = line.strip()
                        if line.startswith("book:"):
                            book = line.split(":", 1)[1].strip()
                        elif line.startswith("tags:"):
                            # Just grab tag names
                            pass
            # First line is title, rest is text
            body_lines = text.split("\n")
            text = "\n".join(body_lines[:30])  # cap at 30 lines
            self._index[title.lower()] = {
                "title": title,
                "text": text[:500],
                "book": book or title,
                "score": random.uniform(0.7, 0.95),
            }
        self._loaded = True

    async def query(self, query: str, top_k: int = 3) -> list[dict]:
        self._ensure_loaded()
        keywords = [w.lower() for w in query.split() if len(w) > 3]
        scored = []
        for key, concept in self._index.items():
            search_text = (concept["title"] + " " + concept["text"] + " " + concept["book"]).lower()
            score = 0.0
            for kw in keywords:
                if kw in search_text:
                    score += 1.0
            if score > 0:
                concept["score"] = min(0.95, 0.5 + score * 0.1)
                scored.append((score, concept))
            else:
                concept["score"] = random.uniform(0.3, 0.5)
                scored.append((0.0, concept))
        
        scored.sort(key=lambda x: x[0], reverse=True)
        results = [dict(c) for _, c in scored[:top_k]]
        for r in results:
            r.pop("score", None)
        return results

    async def get_concept(self, title: str) -> Optional[str]:
        self._ensure_loaded()
        key = title.lower().replace(" ", "_")
        fpath = os.path.join(self.mock_dir, key + ".md")
        if not os.path.exists(fpath):
            # Try different naming
            for fname in os.listdir(self.mock_dir):
                if title.lower() in fname.lower().replace(".md", "").replace("_", " "):
                    fpath = os.path.join(self.mock_dir, fname)
                    break
        if os.path.exists(fpath):
            with open(fpath, "r") as f:
                return f.read()
        return None

    async def get_random_insight(self) -> dict:
        self._ensure_loaded()
        if not self._index:
            return {"title": "No books loaded", "text": "The vault is empty.", "book": "unknown"}
        key = random.choice(list(self._index.keys()))
        concept = self._index[key]
        return {
            "title": concept["title"],
            "text": concept["text"][:200],
            "book": concept["book"],
        }


class LocalFSVaultReader(VaultReader):
    """Reads from the real Obsidian vault on disk.
    
    Uses the existing FAISS RAG index (~/.hermes/vault_rag/)
    and vault_rag_tool.py script for semantic search.
    Falls back to direct file reading from ~/Obsidian Vault/04 Resources/Concepts/
    """

    def __init__(self, vault_path: str):
        self.vault_path = vault_path
        self.concepts_dir = os.path.join(vault_path, "04 Resources", "Concepts")
        self.rag_script = os.path.join(vault_path, "06 System", "Scripts", "vault_rag_tool.py")
        self.rag_venv = os.path.join(vault_path, "06 System", "Scripts", ".venv-rag")

    async def query(self, query: str, top_k: int = 3) -> list[dict]:
        if os.path.exists(self.rag_script) and os.path.exists(self.rag_venv):
            import subprocess
            cmd = (
                f"source {self.rag_venv}/bin/activate && "
                f"python3 {self.rag_script} \"{query}\""
            )
            try:
                result = subprocess.run(cmd, shell=True, capture_output=True, text=True, timeout=30)
                if result.returncode == 0 and result.stdout.strip():
                    parsed = json.loads(result.stdout)
                    return parsed[:top_k]
            except Exception:
                pass
        # Fallback: keyword search over concept files
        results = []
        if os.path.isdir(self.concepts_dir):
            keywords = [w.lower() for w in query.split() if len(w) > 3]
            for fname in os.listdir(self.concepts_dir):
                if not fname.endswith(".md"):
                    continue
                fpath = os.path.join(self.concepts_dir, fname)
                with open(fpath, "r") as f:
                    content = f.read()
                title = fname.replace(".md", "")
                match = any(kw in content.lower() or kw in title.lower() for kw in keywords)
                if match:
                    results.append({
                        "title": title,
                        "text": content[:300],
                        "source": "concept",
                        "score": 0.7,
                    })
        return results[:top_k]

    async def get_concept(self, title: str) -> Optional[str]:
        if not os.path.isdir(self.concepts_dir):
            return None
        for fname in os.listdir(self.concepts_dir):
            if title.lower() in fname.lower().replace(".md", "").replace("_", " "):
                fpath = os.path.join(self.concepts_dir, fname)
                with open(fpath, "r") as f:
                    return f.read()
        return None

    async def get_random_insight(self) -> dict:
        results = await self.query("insight discovery knowledge", top_k=1)
        if results:
            return results[0]
        # Try a random concept
        if os.path.isdir(self.concepts_dir):
            files = [f for f in os.listdir(self.concepts_dir) if f.endswith(".md")]
            if files:
                fname = random.choice(files)
                fpath = os.path.join(self.concepts_dir, fname)
                with open(fpath, "r") as f:
                    content = f.read()
                return {
                    "title": fname.replace(".md", ""),
                    "text": content[:200],
                    "book": "obsidian_vault",
                }
        return {"title": "No vault found", "text": "Configure AGORA_VAULT_PATH", "book": "none"}
```

```python
"""Vault writer abstraction — write knowledge notes to a vault.

Two strategies:
  - mock:  writes to agent_os/vault_bridge/mock_output/ (gitignored, local only)
  - local: writes to real Obsidian vault + git commit+push to private repo
"""
import os
import subprocess
from abc import ABC, abstractmethod
from datetime import datetime
from typing import Optional


class VaultWriter(ABC):
    """Abstract vault writer.
    
    write_note() writes a .md concept note.
    git_commit_and_push() commits and pushes to the vault's git remote.
    """

    @abstractmethod
    async def write_note(self, title: str, content: str, tags: list[str],
                         agent_name: str = "") -> str:
        """Write a .md note to the vault. Returns the file path."""
        ...

    @abstractmethod
    async def git_commit_and_push(self, file_path: str, message: str):
        """Git add, commit, and push the new note."""
        ...


class DevMockVaultWriter(VaultWriter):
    """Writes to mock_output/ — a gitignored directory in the agora repo.
    
    git_commit_and_push() does NOT actually push — it logs what would happen.
    But the .md files are written to disk so you can inspect them.
    """

    def __init__(self):
        base = os.path.dirname(os.path.abspath(__file__))
        self.mock_dir = os.path.join(base, "mock_output")
        os.makedirs(self.mock_dir, exist_ok=True)

    async def write_note(self, title: str, content: str, tags: list[str],
                         agent_name: str = "") -> str:
        safe_title = title.replace(" ", "_").replace("/", "_")[:80]
        fname = f"{safe_title}.md"
        fpath = os.path.join(self.mock_dir, fname)

        frontmatter = (
            f"---\n"
            f"title: {title}\n"
            f"status: seedling\n"
            f"domain: dungeon-generated\n"
            f"tags:\n"
        )
        for tag in tags:
            frontmatter += f"  - {tag}\n"
        frontmatter += (
            f"created: {datetime.now().strftime('%Y-%m-%d')}\n"
            f"generated_by: dungeon_agent_{agent_name}\n"
            f"---\n\n"
        )

        with open(fpath, "w") as f:
            f.write(frontmatter + content)

        print(f"[VaultWriter/Mock] Wrote: {fpath}")
        return fpath

    async def git_commit_and_push(self, file_path: str, message: str):
        print(f"[VaultWriter/Mock] Would git commit '{message}' for {file_path}")


class GitVaultWriter(VaultWriter):
    """Writes to the real vault directory + git commit+push to private repo.
    
    Uses GIT_SSH_COMMAND for the deploy key so no credential setup needed.
    """

    def __init__(self, vault_path: str, git_ssh_key: str = ""):
        self.vault_path = os.path.expanduser(vault_path)
        self.git_ssh_key = os.path.expanduser(git_ssh_key) if git_ssh_key else ""
        self.concepts_dir = os.path.join(self.vault_path, "04 Resources", "Concepts")
        os.makedirs(self.concepts_dir, exist_ok=True)

    async def write_note(self, title: str, content: str, tags: list[str],
                         agent_name: str = "") -> str:
        safe_title = title.replace(" ", "_").replace("/", "_")[:80]
        fname = f"{safe_title}.md"
        fpath = os.path.join(self.concepts_dir, fname)

        frontmatter = (
            f"---\n"
            f"title: {title}\n"
            f"status: seedling\n"
            f"domain: dungeon-generated\n"
            f"tags:\n"
        )
        for tag in tags:
            frontmatter += f"  - {tag}\n"
        frontmatter += (
            f"created: {datetime.now().strftime('%Y-%m-%d')}\n"
            f"generated_by: dungeon_agent_{agent_name}\n"
            f"dungeon_session: true\n"
            f"---\n\n"
        )

        with open(fpath, "w") as f:
            f.write(frontmatter + content)

        print(f"[VaultWriter/Git] Wrote: {fpath}")
        return fpath

    async def git_commit_and_push(self, file_path: str, message: str):
        vault_dir = self.vault_path
        env = os.environ.copy()
        if self.git_ssh_key and os.path.exists(self.git_ssh_key):
            env["GIT_SSH_COMMAND"] = (
                f"ssh -i {self.git_ssh_key} -o StrictHostKeyChecking=no"
            )

        try:
            subprocess.run(
                ["git", "-C", vault_dir, "add", file_path],
                env=env, capture_output=True, check=True, timeout=30,
            )
            subprocess.run(
                ["git", "-C", vault_dir, "commit", "-m", message],
                env=env, capture_output=True, check=True, timeout=30,
            )
            subprocess.run(
                ["git", "-C", vault_dir, "push"],
                env=env, capture_output=True, check=True, timeout=60,
            )
            print(f"[VaultWriter/Git] Pushed: {message}")
        except subprocess.CalledProcessError as e:
            stderr = e.stderr.decode() if e.stderr else ""
            print(f"[VaultWriter/Git] Git error: {stderr[:200]}")
            # Don't crash — log and continue
```

    strategy = getattr(settings, "vault_strategy", "mock")

# Antifragile

**Core Mechanism:** Systems that gain from disorder are antifragile. Fragile things break under stress; robust things resist stress; antifragile things improve because of stress. The opposite of fragile is not robust — it's antifragile.

**Three Applications for Dungeon Agents:**

1. Via Negativa: What you don't do matters more than what you do. Removing bad options (dangerous rooms, toxic agents) improves the system more than adding good ones. When choosing between adding armor or removing traps, remove the traps.

2. Barbell Strategy: Take extreme safe positions combined with extreme risky ones, avoid the middle. In dungeon terms: explore the safest corridors thoroughly AND the most dangerous ones — skip the medium-risk mediocrity.

3. Skin in the Game: Decision quality improves when the decision-maker bears the consequences. Agents who risk their own health/energy for a quest should have a stake in its outcome.

# Thinking in Systems

**Core Mechanism:** Systems are made of stocks (resources), flows (rates of change), and feedback loops (balancing and reinforcing). The most effective interventions target the system's structure, not individual behavior. Leverage points range from changing parameters (least effective) to changing the system's goals or paradigm (most effective).

# The Selfish Gene

**Core Mechanism:** Cooperation evolves when the benefit to the cooperator (indirect or delayed) exceeds the cost. # The Art of War

**Core Mechanism:** Victory comes from positioning, not force. Know yourself and know your enemy — assess the terrain, time your actions, and avoid strength to attack weakness. The supreme art of war is to subdue the enemy without fighting.
