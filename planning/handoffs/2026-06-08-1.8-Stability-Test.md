### Context

This is the **capstone of Phase 1**. Axelrod (1984) proved that Tit-for-Tat is collectively stable — meaning no alternative strategy can invade a population of TFT players. This test proves it empirically with our ESS Framework.

**Experiment design:**

```
Round 1-50:  10 TFT agents, all cooperate → trust builds
Round 51:    Inject 3 defector agents (always defect)
Rounds 52-100: Watch the invasion attempt
Expected:   Defectors get isolated (trust ~0), TFT agents remain stable
```

**Goal:** Create a standalone test script that:
1. Runs the full experiment
2. Prints round-by-round trust metrics
3. Declares PASS/FAIL based on ESS stability criterion
4. Can be run via `python3 -m pytest` or directly

"""
ESS Stability Test — Swarm Invasion Proof

Based on Axelrod's Proposition 2 (1984):
  'TIT FOR TAT is collectively stable if the discount parameter, w,
   is sufficiently large relative to the payoff parameter.'

In our model:
  - TFT agents: always cooperate first, retaliate proportionally, forgive
  - Defector agents: always defect (ALL-D strategy)
  - ESS stability = defectors cannot gain higher trust than TFT agents

Usage:
  pytest tests/test_ess_stability.py -v
  # Or directly:
  python3 -m tests.test_ess_stability
"""

import json
import os
import sys
import time
import uuid
from datetime import datetime, timezone

# Ensure we can import from the project

import pytest

from agora.coordination.ess_protocol import TrustEngine, ESSMessage, MessageType, ESS_TOPICS

# ── Constants ──────────────────────────────────────
TFT_COUNT = 10           # Number of TFT agents in swarm
DEFECTOR_COUNT = 3       # Number of invading defectors
COOPERATION_ROUNDS = 50  # Rounds before invasion
INVASION_ROUNDS = 50     # Rounds after invasion
TOTAL_ROUNDS = COOPERATION_ROUNDS + INVASION_ROUNDS


@pytest.fixture

    conn = await aiosqlite.connect(":memory:")

def make_tft_agent(agent_id: str) -> tuple:
    """Create a TFT agent identity."""
    priv, pub = ESSMessage.generate_keypair()
    return {
        "agent_id": agent_id,
        "role": "tft",
        "strategy": "tft",
        "private_key": priv,
        "public_key": pub,
    }


def make_defector(agent_id: str) -> tuple:
    """Create a defector agent identity."""
    priv, pub = ESSMessage.generate_keypair()
    return {
        "agent_id": agent_id,
        "role": "defector",
        "strategy": "all_d",  # Always defect
        "private_key": priv,
        "public_key": pub,
    }


async def seed_agents(db, agents: list[dict]):
    """Insert agents into the database."""
    for a in agents:
        pub_hex = ESSMessage.public_key_to_hex(a["public_key"])
        genome = json.dumps({"strategy": a["strategy"], "role": a["role"]})
        await db.execute(
            """INSERT OR IGNORE INTO agent_identities
               (agent_id, public_key, generation, genome, trust_score, energy_balance, role, status)
               VALUES (?, ?, 0, ?, 0.5, 100, ?, 'active')""",
            (a["agent_id"], pub_hex, genome, a["role"]),
        )
    await db.commit()


async def run_round(engine: TrustEngine, agents: list[dict], round_num: int) -> dict:
    """Run one round of the experiment.
    
    Each agent interacts with every other agent once.
    TFT agents cooperate first, then mirror partner's last move.
    Defectors always defect.
    """
    metrics = {
        "round": round_num,
        "tft_avg_trust": 0.0,
        "defector_avg_trust": 0.0,
        "tft_tft_trust": 0.0,
        "tft_defector_trust": 0.0,
        "defector_tft_trust": 0.0,
    }

    tft_agents = [a for a in agents if a["strategy"] == "tft"]
    defectors = [a for a in agents if a["strategy"] == "all_d"]
    tft_ids = [a["agent_id"] for a in tft_agents]
    def_ids = [a["agent_id"] for a in defectors]

    # Track last known move for each pair (for TFT mirroring)
    # In a real system this comes from the trust engine — here we simulate
    pair_memory: dict[str, str] = {}

    interactions = 0

    for a in agents:
        for b in agents:
            if a["agent_id"] == b["agent_id"]:
                continue

            # Determine outcome
            if a["strategy"] == "all_d":
                outcome = "defect"
            elif b["strategy"] == "all_d":
                # TFT vs defector: first move cooperate, then mirror
                pair_key = f"{a['agent_id']}:{b['agent_id']}"
                last_move = pair_memory.get(pair_key, "cooperate")
                # This is a's move toward b. If b defected last time,
                # a retaliates. If this is first move, cooperate.
                # Check if b defected against a previously
                reverse_key = f"{b['agent_id']}:{a['agent_id']}"
                b_last = pair_memory.get(reverse_key)
                if b_last == "defect":
                    outcome = "defect"  # Retaliate
                else:
                    outcome = "cooperate"  # Nice — never defect first
            else:
                # TFT vs TFT: cooperate always (both are nice)
                outcome = "cooperate"

            # Record interaction in trust engine
            await engine.record_interaction(a["agent_id"], b["agent_id"], outcome)
            pair_memory[f"{a['agent_id']}:{b['agent_id']}"] = outcome
            interactions += 1

    # Compute metrics
    tft_trusts = []
    def_trusts = []
    tft_tft_trusts = []
    tft_def_trusts = []
    def_tft_trusts = []

    for tid in tft_ids:
        for pid in tft_ids:
            if tid != pid:
                t = await engine.get_trust(tid, pid)
                tft_trusts.append(t)
                tft_tft_trusts.append(t)

        for did in def_ids:
            t1 = await engine.get_trust(tid, did)
            t2 = await engine.get_trust(did, tid)
            tft_trusts.append(t1)
            def_trusts.append(t2)
            tft_def_trusts.append(t1)
            def_tft_trusts.append(t2)

    for did in def_ids:
        for pid2 in tft_ids:
            t = await engine.get_trust(did, pid2)
            def_trusts.append(t)

    metrics["tft_avg_trust"] = round(sum(tft_trusts) / max(len(tft_trusts), 1), 4)
    metrics["defector_avg_trust"] = round(sum(def_trusts) / max(len(def_trusts), 1), 4) if def_trusts else 0.0
    metrics["tft_tft_trust"] = round(sum(tft_tft_trusts) / max(len(tft_tft_trusts), 1), 4) if tft_tft_trusts else 0.0
    metrics["tft_defector_trust"] = round(sum(tft_def_trusts) / max(len(tft_def_trusts), 1), 4) if tft_def_trusts else 0.0
    metrics["defector_tft_trust"] = round(sum(def_tft_trusts) / max(len(def_tft_trusts), 1), 4) if def_tft_trusts else 0.0
    metrics["interactions"] = interactions

    return metrics


def print_report(all_metrics: list[dict]):
    """Print a human-readable experiment report."""
    print("\n" + "=" * 70)
    print("  ESS STABILITY TEST — Swarm Invasion Report")
    print("=" * 70)
    print(f"\n  Swarm: {TFT_COUNT} TFT agents + {DEFECTOR_COUNT} defectors")
    print(f"  Cooperation rounds: {COOPERATION_ROUNDS}")
    print(f"  Invasion rounds:    {INVASION_ROUNDS}")
    print()

    # Summary stats
    pre = [m for m in all_metrics if m["round"] <= COOPERATION_ROUNDS]
    post = [m for m in all_metrics if m["round"] > COOPERATION_ROUNDS]

    pre_tft = [m["tft_avg_trust"] for m in pre]
    post_tft = [m["tft_avg_trust"] for m in post]
    post_def = [m["defector_avg_trust"] for m in post]

    pre_tft_tft = [m["tft_tft_trust"] for m in pre]
    post_tft_tft = [m["tft_tft_trust"] for m in post]
    post_tft_def = [m["tft_defector_trust"] for m in post]

    print(f"  📊 Pre-invasion (rounds 1-{COOPERATION_ROUNDS}):")
    print(f"     TFT↔TFT avg trust:   {sum(pre_tft_tft)/len(pre_tft_tft):.4f}")
    print(f"     All agents avg trust: {sum(pre_tft)/len(pre_tft):.4f}")
    print()

    print(f"  📊 Post-invasion (rounds {COOPERATION_ROUNDS+1}-{TOTAL_ROUNDS}):")
    print(f"     TFT↔TFT avg trust:   {sum(post_tft_tft)/len(post_tft_tft):.4f}")
    print(f"     TFT→Defector trust:  {sum(post_tft_def)/len(post_tft_def):.4f}")
    print(f"     Defector→TFT trust:  {sum(m['defector_tft_trust'] for m in post)/len(post):.4f}")
    print(f"     Defector avg trust:  {sum(post_def)/len(post_def):.4f}")
    print()

    # ESS stability criterion
    tft_stable = sum(post_tft_tft) / len(post_tft_tft) > 0.3  # Above baseline
    defector_rejected = sum(post_tft_def) / len(post_tft_def) < 0.2  # Below threshold

    print(f"  🧪 ESS Stability Test Result:")
    if tft_stable and defector_rejected:
        print(f"     ✅ PASS — TFT population is ESS")
        print(f"        TFT trust preserved ({sum(post_tft_tft)/len(post_tft_tft):.4f})")
        print(f"        Defectors rejected ({sum(post_tft_def)/len(post_tft_def):.4f})")
        print(f"        → Axelrod's Proposition 2 confirmed")
    elif tft_stable and not defector_rejected:
        print(f"     ⚠️ PARTIAL — TFT stable but defectors not isolated")
        print(f"        Check forgiveness rate or baseline trust")
    else:
        print(f"     ❌ FAIL — TFT population destabilized by invasion")
        print(f"        Check TFT implementation")

    # Round-by-round snapshot (first 5, invasion point, last 5)
    print(f"\n  📈 Round-by-round (key frames):")
    print(f"     {'Round':>6} | {'TFT-TFT':>8} | {'TFT-Def':>8} | {'Def-TFT':>8} | {'Def avg':>8}")
    print(f"     {'-'*6}-+-{'-'*8}-+-{'-'*8}-+-{'-'*8}-+-{'-'*8}")

    key_rounds = (
        [1] +
        [COOPERATION_ROUNDS // 2] +
        [COOPERATION_ROUNDS] +
        [COOPERATION_ROUNDS + 3] +
        [COOPERATION_ROUNDS + INVASION_ROUNDS // 2] +
        [TOTAL_ROUNDS]
    )
    for m in all_metrics:
        if m["round"] in key_rounds:
            print(f"     {m['round']:>6} | {m['tft_tft_trust']:>8.4f} | {m['tft_defector_trust']:>8.4f} | {m['defector_tft_trust']:>8.4f} | {m['defector_avg_trust']:>8.4f}")

    print("\n" + "=" * 70)
    print()


# ── Pytest Test ────────────────────────────────────

@pytest.mark.asyncio
async def test_ess_stability(db):
    """Run the ESS stability experiment and assert TFT is collectively stable."""
    engine = TrustEngine(db)

    # Create agents
    tft_agents = [make_tft_agent(f"tft-{i:03d}") for i in range(TFT_COUNT)]
    defectors = [make_defector(f"def-{i:03d}") for i in range(DEFECTOR_COUNT)]
    all_agents = tft_agents + defectors

    # Seed into DB
    await seed_agents(db, all_agents)

    # Run experiment
    all_metrics = []
    for rnd in range(1, TOTAL_ROUNDS + 1):
        metrics = await run_round(engine, all_agents, rnd)
        all_metrics.append(metrics)

    # Print report
    print_report(all_metrics)

    # Assertions
    post = [m for m in all_metrics if m["round"] > COOPERATION_ROUNDS]
    tft_stable = sum(m["tft_tft_trust"] for m in post) / len(post)
    defector_trust = sum(m["tft_defector_trust"] for m in post) / len(post)

    assert tft_stable > 0.3, f"TFT trust collapsed: {tft_stable:.4f}"
    assert defector_trust < 0.25, f"Defectors not rejected: {defector_trust:.4f}"

    print(f"  ✅ All assertions passed — TFT is ESS")
