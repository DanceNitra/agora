There are **4 existing subsystems** that don't talk to each other:

| Subsystem | File | Status |
|-----------|------|--------|
| 🧠 **AgentOS** | `agent_os/agent_os.py` (888L) | 7 NPCs with soul/brain/body/abilities/skills. `_think()` calls LLM but very shallow. |
| 🗣️ **LLM Client** | `execution/llm_client.py` (465L) | `dungeon_agent_think()` calls Nemotron 3 Ultra 550B via OpenRouter. **FREE.** |
| 🔗 **ESS Trust** | `coordination/` | TrustEngine + Stigmergy + TFT — but NOT fed into NPC decisions yet. |
| 🎮 **Dungeon Visual** | MCP tools + shell/game | Agents rendered as GLTF characters, move/state/thought/health. |

**The gap:** Each NPC calls the LLM to decide their state_of_mind, but:
1. They don't know their **trust score** with other NPCs (ESS data)
2. They don't know what **other agents are doing** (stigmergy traces)
3. Their **skills/abilities** are stored in DB but never used in LLM prompts
4. LLM decisions never trigger **visible actions** (move, emote, cooperate)
5. **Help-seeking** is rule-based, not LLM-driven

**Goal:** One integrated tick where each NPC:
1. Reads their ESS trust scores + stigmergy traces + own skills
2. Calls Nemotron with a rich character prompt 
3. LLM decides: cooperate/defect/move/seek-help/share
4. Decision executes visibly in dungeon AND records in ESS

### Step 2 — Execute LLM decisions visibly — `agent_os/agent_os.py`

After `_think()` returns the LLM decision (line 440-453), add an **execution phase**. Add this as a new method `_execute_llm_decision()`:

```python
async def _execute_llm_decision(
    self,
    npc_id: str,
    name: str,
    decision: dict,
    npc: dict,
    broadcast_fn=None,
    engine=None,
):
    """Execute an LLM decision visibly in the dungeon and record in ESS."""
    action = decision.get("action", "explore")
    target_id = decision.get("target_id")
    insight = decision.get("insight", "")
    skill_to_use = decision.get("skill_to_use")
    new_state = decision.get("state_of_mind", "focused")

    # ── Update state_of_mind in brain ──
    await self.state_store.update_brain(npc_id, {
        "state_of_mind": new_state,
        "last_decision": insight,
    })

    # ── Visible dungeon actions ──
    if action == "move_to" and target_id:
        # Move toward target (position update)
        target_npc = await self.state_store.get_npc(target_id)
        if target_npc:
            tx, ty = target_npc.get("pos_x", 320), target_npc.get("pos_y", 240)
            nx, ny = npc.get("pos_x", 320), npc.get("pos_y", 240)
            # Move 20px closer per tick (smooth)
            dx = (tx - nx) / max(abs(tx - nx), 1) * 20
            dy = (ty - ny) / max(abs(ty - ny), 1) * 20
            new_x = nx + dx
            new_y = ny + dy
            await self.state_store.update_npc(npc_id, {"pos_x": new_x, "pos_y": new_y})

    elif action in ("cooperate", "share") and target_id:
        # Record ESS interaction as cooperate (only if engine available)
        if engine:
            await engine.record_interaction(npc_id, target_id, "cooperate")

        # Health boost for the target (cooperation reward)
        target_npc = await self.state_store.get_npc(target_id)
        if target_npc:
            new_health = min(100, target_npc.get("health", 100) + 2)
            await self.state_store.update_npc(target_id, {"health": new_health})

    elif action == "defect" and target_id:
        # Record ESS interaction as defect
        if engine:
            await engine.record_interaction(npc_id, target_id, "defect")

    elif action == "seek_help":
        # Trigger help-seeking with LLM-chosen problem type
        problem_type = self._classify_problem(decision.get("goal", ""), name)
        await self._seek_help_auto(npc_id, name, broadcast_fn)

    elif action == "rest":
        # Recover stamina
        body = await self.state_store.get_body(npc_id)
        if body:
            new_stamina = min(100, body.get("stamina", 50) + 15)
            await self.state_store.update_body(npc_id, {"stamina": new_stamina})

    # ── Skill XP gain ──
    if skill_to_use:
        await self.db.execute(
            "UPDATE agent_skills SET xp = xp + 5, last_used_at = datetime('now') "
            "WHERE npc_id=? AND skill_name=?",
            (npc_id, skill_to_use),
        )

    # ── Broadcast thought bubble ──
    if broadcast_fn and insight:
        thought_data = {
            "agent_id": name,
            "action": action,
            "insight": insight[:120],
            "state_of_mind": new_state,
            "target_id": target_id,
        }
        if hasattr(broadcast_fn, '__call__'):
            await broadcast_fn("agent_thought", thought_data)
        elif isinstance(broadcast_fn, dict) and "event_bus" in broadcast_fn:
            await broadcast_fn["event_bus"].publish("agent:events", "agent_thought", thought_data)
```

### Step 3 — Wire the integrated tick — `server/agora/main.py`

```python
async def _integrated_tick(app: FastAPI):
    """One tick: AgentOS think → LLM decide → ESS record → Dungeon broadcast."""
    db = app.state.db
    event_bus = getattr(app.state, "event_bus", None)
    engine = getattr(app.state, "trust", None)
    stigmergy = getattr(app.state, "stigmergy", None)
    agent_os = getattr(app.state, "agent_os", None)

    if not agent_os or not db:
        return

    # Get all active NPCs
    cursor = await db.execute(
        "SELECT d.npc_id, d.npc_name, d.role, d.health, d.status, "
        "d.pos_x, d.pos_y, b.state_of_mind "
        "FROM dungeon_npcs d "
        "LEFT JOIN agent_brain b ON b.npc_id = d.npc_id "
        "WHERE d.status='active'"
    )
    npcs = await cursor.fetchall()

    for npc in npcs:
        npc_id = npc["npc_id"]
        name = npc["npc_name"]

        # 1. Body update
        await agent_os._update_body(npc_id)

        # 2. Think (LLM) — already enhanced with skills/trust/stigmergy
        state_of_mind = await agent_os._think(npc_id, name)

        # If LLM was used, _think already updated brain.
        # If LLM failed, _think_rule_based was used.

        # 3. If confused/panicked → seek help
        if state_of_mind in ("confused", "panicked"):
            await agent_os._seek_help_auto(npc_id, name, broadcast_fn={
                "event_bus": event_bus,
            })

        # 4. Write stigmergy trace
        if stigmergy:
            await stigmergy.write_trace(
                agent_id=npc_id,
                task_type=npc["role"],
                result=f"State: {state_of_mind}",
                trust_delta=0.0,
            )

    # 5. Epoch engine tick
    epoch_engine = getattr(app.state, "epoch_engine", None)
    if epoch_engine:
        await epoch_engine.tick(app)

    # 6. Checkpoint every N ticks
    checkpointer = getattr(app.state, "checkpointer", None)
    if checkpointer and app.state.tick_count % 50 == 0:
        await checkpointer.checkpoint_all("trust")
        await checkpointer.checkpoint_all("tft")

    app.state.tick_count += 1
```
